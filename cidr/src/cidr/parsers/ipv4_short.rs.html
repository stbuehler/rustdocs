16// parse normal IPv4 addresses as host cidr, and short forms
17// with a cidr that indicates how many octets were present.
18fn _parse_short_ipv4_address_as_cidr(s: &str) -> Option<Ipv4Cidr> {
19	let mut octets = [0u8; 4];
20	let mut last = 0;
21	for (ndx, os) in s.split('.').enumerate() {
22		if ndx >= 4 {
23			// too many octets
24			return None;
25		}
26		// abort on invalid octet
27		octets[ndx] = os.parse().ok()?;
28		last = ndx;
29	}
30	let bits = (last as u8 + 1) * 8;
31	Some(Ipv4Cidr::new(Ipv4Addr::from(octets), bits).expect("host bits are zero"))
32}

34/// Parse "short" IPv4 addresses as networks with octet-aligned length
35///
36/// * parse `"10"` as `10.0.0.0/8`
37/// * parse `"192.168"` as `192.168.0.0/16`
38/// * parse `"192.0.2"` as `192.0.2.0/24`
39/// * parse `"127"` as `127.0.0.0/8`
40/// * parse `"127.0.0.1"` as `127.0.0.1/32`
41///
42/// The returned prefix length indicates how many octets were present.
43///
44/// This is very different from [`inet_addr`][`super::inet_addr`] which would
45/// interpret `"192.168"` as `192.0.0.168`!
46///
47/// This function doesn't accept normal CIDR notations, so you will probably
48/// need to combine it with other functions.
49pub fn parse_short_ipv4_address_as_cidr(s: &str) -> Result<Ipv4Cidr, NetworkParseError> {
50	match _parse_short_ipv4_address_as_cidr(s) {
51		Some(n) => Ok(n),
52		None => {
53			// address parser should fail here (to generate proper error result)
54			// (but if it works the result should actually be correct)
55			Ok(Ipv4Cidr::new_host(s.parse()?))
56		},
57	}
58}

60/// Parses normal IPv4 CIDR notations or short forms via [`parse_short_ipv4_address_as_cidr`]
61pub fn parse_short_ipv4_cidr(s: &str) -> Result<Ipv4Cidr, NetworkParseError> {
62	super::parse_cidr_full(s, FromStr::from_str, parse_short_ipv4_address_as_cidr)

65/// Parses normal IP addresses as host addresses, and short IPv4 addresses via [`parse_short_ipv4_address_as_cidr`]
66///
67/// This function doesn't accept normal CIDR notations, so you will probably
68/// need to combine it with other functions.
69pub fn parse_short_ip_address_as_cidr(s: &str) -> Result<IpCidr, NetworkParseError> {
70	match s.parse::<IpAddr>() {
71		Ok(a) => Ok(IpCidr::new_host(a)),
72		// only try short IPv4 as fallback
73		Err(e) => match _parse_short_ipv4_address_as_cidr(s) {
74			Some(n) => Ok(n.into()),
75			None => Err(e.into()),
76		},
77	}
78}
