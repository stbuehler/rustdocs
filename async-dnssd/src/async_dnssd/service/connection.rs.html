56/// Pending record registration
57///
58/// Becomes invalid when the future completes; use the returned
59/// [`Record`](struct.Record.html) instead.
60// the future gets canceled by dropping the record; must
61// not drop the future without dropping the record.
62#[must_use = "futures do nothing unless polled"]
63pub struct RegisterRecord {
64	future: CallbackFuture,
65	record: Option<crate::Record>,
66}
67
68impl Future for RegisterRecord {
69	type Output = io::Result<crate::Record>;
70
71	fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
72		let this = self.get_mut();
73		futures_core::ready!(this.future.poll_unpin(cx))?;
74		Poll::Ready(Ok(this.record.take().unwrap()))
75	}
76}

93/// Optional data when registering a record; either use its default
94/// value or customize it like:
95///
96/// ```
97/// # use async_dnssd::RegisterRecordData;
98/// RegisterRecordData {
99///     ttl: 60,
100///     ..Default::default()
101/// };
102/// ```
103#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Debug)]
104pub struct RegisterRecordData {
105	/// flags for registration
106	pub flags: RegisterRecordFlags,
107	/// interface to register record on
108	pub interface: Interface,
109	/// class of the resource record (default: `IN`)
110	pub rr_class: Class,
111	/// time to live of the resource record in seconds (passing 0 will
112	/// select a sensible default)
113	pub ttl: u32,
114	#[doc(hidden)]
115	pub _non_exhaustive: crate::non_exhaustive_struct::NonExhaustiveMarker,
116}
117
118impl Default for RegisterRecordData {
119	fn default() -> Self {
120		Self {
121			flags: RegisterRecordFlags::default(),
122			interface: Interface::default(),
123			rr_class: Class::IN,
124			ttl: 0,
125			_non_exhaustive: crate::non_exhaustive_struct::NonExhaustiveMarker,
126		}
127	}
128}

130impl Connection {
131	/// Register record on interface with given name, type, class, rdata
132	/// and ttl
133	///
134	/// See [`DNSServiceRegisterRecord`](https://developer.apple.com/documentation/dnssd/1804727-dnsserviceregisterrecord).
135	#[doc(alias = "DNSServiceRegisterRecord")]
136	pub fn register_record_extended(
137		&self,
138		fullname: &str,
139		rr_type: Type,
140		rdata: &[u8],
141		data: RegisterRecordData,
142	) -> io::Result<RegisterRecord> {
143		let fullname = cstr::CStr::from(&fullname)?;
144
145		let (future, record) = CallbackFuture::new_with(self.0.clone(), move |sender| {
146			self.0.clone().register_record(
147				data.flags.bits(),
148				data.interface.into_raw(),
149				&fullname,
150				rr_type,
151				data.rr_class,
152				rdata,
153				data.ttl,
154				Some(register_record_callback),
155				sender,
156			)
157		})?;
158
159		Ok(RegisterRecord {
160			future,
161			record: Some(record.into()),
162		})
163	}
164
165	/// Register record on interface with given name, type, class, rdata
166	/// and ttl
167	///
168	/// Uses [`register_record_extended`] with default [`RegisterRecordData`].
169	///
170	/// See [`DNSServiceRegisterRecord`](https://developer.apple.com/documentation/dnssd/1804727-dnsserviceregisterrecord).
171	///
172	/// [`register_record_extended`]: fn.register_record_extended.html
173	/// [`RegisterRecordData`]: struct.RegisterRecordData.html
174	#[doc(alias = "DNSServiceRegisterRecord")]
175	pub fn register_record(
176		&self,
177		fullname: &str,
178		rr_type: Type,
179		rdata: &[u8],
180	) -> io::Result<RegisterRecord> {
181		self.register_record_extended(fullname, rr_type, rdata, RegisterRecordData::default())
182	}
183}

185impl RegisterRecord {
186	fn inner_record(&self) -> &crate::Record {
187		self.record.as_ref().expect("RegisterRecord future is done")
188	}
189
190	/// Type of the record
191	///
192	/// # Panics
193	///
194	/// Panics after the future completed.  Use the returned
195	/// [`Record`](struct.Record.html) instead.
196	pub fn rr_type(&self) -> Type {
197		self.inner_record().rr_type()
198	}
199
200	/// Update record
201	///
202	/// Cannot change type or class of record.
203	///
204	/// # Panics
205	///
206	/// Panics after the future completed.  Use the returned
207	/// [`Record`](struct.Record.html) instead.
208	///
209	/// See [`DNSServiceUpdateRecord`](https://developer.apple.com/documentation/dnssd/1804739-dnsserviceupdaterecord).
210	#[doc(alias = "DNSServiceUpdateRecord")]
211	pub fn update_record(&self, rdata: &[u8], ttl: u32) -> io::Result<()> {
212		self.inner_record().update_record(rdata, ttl)
213	}
214
215	/// Keep record for as long as the underlying connection lives.
216	///
217	/// Keep the a handle to the underlying connection (either the
218	/// [`Connection`](struct.Connection.html) or some other record from
219	/// the same `Connection`) alive.
220	///
221	/// Due to some implementation detail the underlying connection
222	/// might live until this future successfully completes.
223	///
224	/// # Panics
225	///
226	/// Panics after the future completed.  Use the returned
227	/// [`Record`](struct.Record.html) instead.
228	// - implementation detail: this drives the future to continuation,
229	//   it is not possible to drop the (shared) underlying service
230	//   before. instead we could store the callback context with the
231	//   underyling service, and drop it either when dropping the
232	//   service or the callback was called.
233	pub fn keep(self) {
234		let (fut, rec) = (
235			self.future,
236			self.record.expect("RegisterRecord future is done"),
237		);
238		// drive future to continuation, ignore errors
239		tokio::spawn(fut.map(|_| ()));
240		rec.keep();
241	}
